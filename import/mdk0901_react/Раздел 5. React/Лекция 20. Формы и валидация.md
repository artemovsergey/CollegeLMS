### **1. Управляемые формы**

```tsx
function LoginForm() {
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    console.log({ email, password });
  };

  return (
    <form onSubmit={handleSubmit}>
      <input
        type="email"
        value={email}
        onChange={e => setEmail(e.target.value)}
        placeholder="Email"
      />
      <input
        type="password"
        value={password}
        onChange={e => setPassword(e.target.value)}
        placeholder="Пароль"
      />
      <button type="submit">Войти</button>
    </form>
  );
}
```

Каждое поле — управляемый компонент: значение из state, изменение через `onChange`.

### **2. Валидация**

Простая проверка перед отправкой:

```tsx
const [errors, setErrors] = useState<Record<string, string>>({});

const validate = () => {
  const next: Record<string, string> = {};
  if (email.trim().length < 3) next.email = "Введите email";
  if (password.length < 6) next.password = "Минимум 6 символов";
  setErrors(next);
  return Object.keys(next).length === 0;
};
```

**React Hook Form + Zod** — популярная связка для валидации:

```tsx
import { useForm } from "react-hook-form";
import { z } from "zod";
import { zodResolver } from "@hookform/resolvers/zod";

const schema = z.object({
  email: z.string().email("Некорректный email"),
  password: z.string().min(6, "Минимум 6 символов"),
});

function LoginForm() {
  const { register, handleSubmit, formState: { errors } } = useForm({
    resolver: zodResolver(schema),
  });

  const onSubmit = (data: unknown) => console.log(data);

  return (
    <form onSubmit={handleSubmit(onSubmit)}>
      <input {...register("email")} />
      {errors.email && <p className="error">{String(errors.email.message)}</p>}
      <input {...register("password")} type="password" />
      {errors.password && <p className="error">{String(errors.password.message)}</p>}
      <button type="submit">Войти</button>
    </form>
  );
}
```

### **3. Работа с сервером**

```tsx
const [loading, setLoading] = useState(false);

const handleSubmit = async (e: React.FormEvent) => {
  e.preventDefault();
  if (!validate()) return;
  setLoading(true);
  try {
    const res = await fetch("/api/login", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ email, password }),
    });
    if (!res.ok) throw new Error("Ошибка входа");
    alert("Успешно!");
  } catch (err) {
    alert(err.message);
  } finally {
    setLoading(false);
  }
};
```

Не забывайте: кнопка отправки должна блокироваться во время запроса (`disabled={loading}`), а ошибки сервера — показываться пользователю.
