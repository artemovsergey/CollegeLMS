### **1. React Router**

```tsx
import { BrowserRouter, Routes, Route, Link, useParams } from "react-router-dom";

function App() {
  return (
    <BrowserRouter>
      <nav>
        <Link to="/">Главная</Link>
        <Link to="/users">Пользователи</Link>
      </nav>
      <Routes>
        <Route path="/" element={<Home />} />
        <Route path="/users" element={<UsersPage />} />
        <Route path="/users/:id" element={<UserPage />} />
        <Route path="*" element={<NotFound />} />
      </Routes>
    </BrowserRouter>
  );
}

function UserPage() {
  const { id } = useParams();
  return <p>Пользователь {id}</p>;
}
```

Переходы программно:

```tsx
const navigate = useNavigate();
navigate("/users");
```

### **2. Защита маршрутов (Guards)**

Компонент-обёртка проверяет авторизацию и роль:

```tsx
function ProtectedRoute({ role, children }: { role?: string; children: React.ReactNode }) {
  const user = useAuth();

  if (!user) {
    return <Navigate to="/login" replace />;
  }
  if (role && user.role !== role) {
    return <Navigate to="/" replace />;
  }
  return children;
}

// Использование
<Route
  path="/admin"
  element={
    <ProtectedRoute role="Admin">
      <AdminPage />
    </ProtectedRoute>
  }
/>
```

Хук авторизации (упрощённо):

```tsx
function useAuth() {
  const user = JSON.parse(localStorage.getItem("user") || "null");
  return user;
}
```

### **3. Динамические маршруты и 404**

- `:param` — динамический сегмент (доступен через `useParams`).
- Вложенность: `<Route path="settings" element={...} />` внутри родителя с `<Outlet />`.
- `path="*"` — перехват неизвестных путей (страница 404).
