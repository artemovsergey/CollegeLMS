### **1. fetch API**

```tsx
async function getCourses() {
  const res = await fetch("/api/courses", {
    headers: { Authorization: `Bearer ${token}` },
  });
  if (!res.ok) throw new Error(`HTTP ${res.status}`);
  return res.json();
}
```

**Важно:** `fetch` не считает 404/500 ошибкой — статус нужно проверять через `res.ok`.

### **2. Axios**

```tsx
import axios from "axios";

const api = axios.create({
  baseURL: "/api",
  headers: { "Content-Type": "application/json" },
});
```

**Типизированные ответы:**

```tsx
type Course = { id: string; title: string };

async function loadCourses(): Promise<Course[]> {
  const { data } = await api.get<Course[]>("/courses");
  return data;
}
```

**Перехватчики** — общая обработка токенов и ошибок:

```tsx
api.interceptors.request.use(config => {
  const token = localStorage.getItem("token");
  if (token) config.headers.Authorization = `Bearer ${token}`;
  return config;
});

api.interceptors.response.use(
  res => res,
  error => {
    if (error.response?.status === 401) {
      localStorage.removeItem("token");
      window.location.href = "/login";
    }
    return Promise.reject(error);
  }
);
```

### **3. Загрузка данных в компоненте**

```tsx
function CoursesPage() {
  const [courses, setCourses] = useState<Course[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    const controller = new AbortController();

    loadCourses(controller.signal)
      .then(setCourses)
      .catch(e => setError(e.message))
      .finally(() => setLoading(false));

    return () => controller.abort();
  }, []);

  if (loading) return <p>Загрузка...</p>;
  if (error) return <p className="error">Ошибка: {error}</p>;
  return <ul>{courses.map(c => <li key={c.id}>{c.title}</li>)}</ul>;
}
```

**Паттерн трёх состояний** (`loading` / `error` / `data`) — базовый для всех страниц с данными.
